# Extract SYN packets (reconnaissance)
tshark -r capture.pcapng -Y "ip.src==192.168.1.100 && tcp.flags.syn==1 && tcp.flags.ack==0" -T fields -e frame.number -e ip.src -e ip.dst -e tcp.dstport

# Extract HTTP POST requests (exploitation phase)
tshark -r capture.pcapng -Y "http.request.method == POST" -T fields -e frame.number -e ip.src -e http.host -e http.request.uri

# Extract DNS queries (possible C2 communication)
tshark -r capture.pcapng -Y "dns.flags.response == 0" -T fields -e frame.number -e ip.src -e dns.qry.name
